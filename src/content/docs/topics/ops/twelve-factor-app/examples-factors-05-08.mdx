## Factor V: Build, release, run

The build creates one immutable artifact. A release adds deployment configuration and an identity. Runtime selects a process command without downloading dependencies or compiling source.

<Tabs syncKey="twelve-factor-language">
  <TabItem label="TypeScript">
    A multi-stage container restores from the npm lock, compiles once, and copies only runtime files forward. The source revision labels the artifact while deploy configuration stays outside it.

```dockerfile
FROM node:24-bookworm-slim AS build
WORKDIR /app
COPY package.json package-lock.json ./
RUN npm ci
COPY tsconfig.json ./
COPY src ./src
RUN npm run build && npm prune --omit=dev

FROM node:24-bookworm-slim AS runtime
ARG SOURCE_REVISION
LABEL org.opencontainers.image.revision=$SOURCE_REVISION
ENV NODE_ENV=production
WORKDIR /app
COPY --from=build /app/package.json ./package.json
COPY --from=build /app/node_modules ./node_modules
COPY --from=build /app/dist ./dist
USER node
CMD ["node", "dist/web.js"]
```
  </TabItem>
  <TabItem label="Python">
    uv validates the lock and builds one wheel. The runtime installs the locked dependency set and project wheel before the artifact is published.

```dockerfile
FROM python:3.13-slim AS build
WORKDIR /app
RUN pip install --no-cache-dir uv
COPY pyproject.toml uv.lock ./
RUN uv export --frozen --no-dev --no-emit-project --output-file requirements.lock
COPY src ./src
RUN uv build --wheel

FROM python:3.13-slim AS runtime
ARG SOURCE_REVISION
LABEL org.opencontainers.image.revision=$SOURCE_REVISION
WORKDIR /app
COPY --from=build /app/requirements.lock ./requirements.lock
COPY --from=build /app/dist /wheels
RUN pip install --no-cache-dir -r requirements.lock \
    && pip install --no-cache-dir --no-deps /wheels/*.whl \
    && rm -rf requirements.lock /wheels
USER nobody
CMD ["order-web"]
```
  </TabItem>
  <TabItem label="Go">
    The build embeds the source revision in one binary. The runtime image copies those bytes and contains no Go toolchain or module cache.

```dockerfile
FROM golang:1.26-bookworm AS build
WORKDIR /src
COPY go.mod go.sum ./
RUN go mod download && go mod verify
COPY cmd ./cmd
COPY internal ./internal
ARG SOURCE_REVISION
RUN CGO_ENABLED=0 go build \
    -ldflags "-s -w -X main.revision=$SOURCE_REVISION" \
    -o /out/order-service ./cmd/order-service

FROM gcr.io/distroless/static-debian12:nonroot
ARG SOURCE_REVISION
LABEL org.opencontainers.image.revision=$SOURCE_REVISION
COPY --from=build /out/order-service /order-service
ENTRYPOINT ["/order-service"]
CMD ["web"]
```
  </TabItem>
</Tabs>

**Evidence to collect:** Promote the same artifact digest between environments, change only release configuration, and roll back without rebuilding.

## Factor VI: Processes

Web and worker instances keep no required durable state in memory or on their local filesystem. PostgreSQL stores orders and idempotency records. Redis coordinates queued work but does not replace the system of record.

<Tabs syncKey="twelve-factor-language">
  <TabItem label="TypeScript">
    The Fastify route receives a PostgreSQL pool. The database generates and persists the order identifier, so a later request can reach any web process.

```typescript
import type { FastifyInstance } from "fastify";
import type { Pool } from "pg";
import { z } from "zod";

const CreateOrder = z.object({ sku: z.string().min(1), quantity: z.number().int().positive() });

export function registerOrderRoutes(app: FastifyInstance, database: Pool): void {
  app.post("/orders", async (request, reply) => {
    const order = CreateOrder.parse(request.body);
    const result = await database.query(
      `INSERT INTO orders (sku, quantity)
       VALUES ($1, $2)
       RETURNING id, sku, quantity, status`,
      [order.sku, order.quantity],
    );
    return reply.code(201).send(result.rows[0]);
  });
}
```
  </TabItem>
  <TabItem label="Python">
    FastAPI retrieves the pool from application state and writes through Psycopg. Application state holds a connection resource, not the durable order.

```python
from typing import Annotated

from fastapi import Depends, FastAPI, Request, status
from pydantic import BaseModel, Field
from psycopg.rows import dict_row
from psycopg_pool import AsyncConnectionPool

app = FastAPI()


class CreateOrder(BaseModel):
    sku: str = Field(min_length=1)
    quantity: int = Field(gt=0)


def get_database(request: Request) -> AsyncConnectionPool:
    return request.app.state.database


@app.post("/orders", status_code=status.HTTP_201_CREATED)
async def create_order(
    order: CreateOrder,
    database: Annotated[AsyncConnectionPool, Depends(get_database)],
) -> dict[str, object]:
    async with database.connection() as connection:
        async with connection.cursor(row_factory=dict_row) as cursor:
            await cursor.execute(
                "INSERT INTO orders (sku, quantity) VALUES (%s, %s) "
                "RETURNING id, sku, quantity, status",
                (order.sku, order.quantity),
            )
            row = await cursor.fetchone()
    if row is None:
        raise RuntimeError("database did not return the created order")
    return row
```
  </TabItem>
  <TabItem label="Go">
    A chi handler owns no durable map. Its `pgxpool` dependency writes the order and returns the stored representation.

```go
package httpapi

import (
    "encoding/json"
    "net/http"

    "github.com/go-chi/chi/v5"
    "github.com/jackc/pgx/v5/pgxpool"
)

type Server struct {
    database *pgxpool.Pool
}

func New(database *pgxpool.Pool) http.Handler {
    server := &Server{database: database}
    router := chi.NewRouter()
    router.Post("/orders", server.createOrder)
    return router
}

func (s *Server) createOrder(writer http.ResponseWriter, request *http.Request) {
    var input struct {
        SKU      string `json:"sku"`
        Quantity int    `json:"quantity"`
    }
    if err := json.NewDecoder(request.Body).Decode(&input); err != nil {
        http.Error(writer, "invalid request", http.StatusBadRequest)
        return
    }

    var order struct {
        ID     string `json:"id"`
        Status string `json:"status"`
    }
    err := s.database.QueryRow(
        request.Context(),
        `INSERT INTO orders (sku, quantity)
         VALUES ($1, $2) RETURNING id, status`,
        input.SKU,
        input.Quantity,
    ).Scan(&order.ID, &order.Status)
    if err != nil {
        http.Error(writer, "order storage failed", http.StatusInternalServerError)
        return
    }
    writer.WriteHeader(http.StatusCreated)
    _ = json.NewEncoder(writer).Encode(order)
}
```
  </TabItem>
</Tabs>

**Evidence to collect:** Terminate the instance that accepted an order, route the next read to another instance, and verify the same durable result.

## Factor VII: Port binding

The web process owns its HTTP listener and binds the configured port. The platform routes to that listener. Workers and admin jobs expose commands because their invocation model is not HTTP.

<Tabs syncKey="twelve-factor-language">
  <TabItem label="TypeScript">
    Fastify owns the HTTP server and listens on the validated port. Rejected startup propagates to the process entry point, so a bind failure cannot look ready.

```typescript
import Fastify from "fastify";

import { config } from "./config.js";
import { registerOrderRoutes } from "./orders.js";
import { database } from "./resources.js";

const app = Fastify({ logger: true });
registerOrderRoutes(app, database);

app.get("/live", async () => ({ status: "alive" }));

await app.listen({
  host: "0.0.0.0",
  port: config.PORT,
});
```
  </TabItem>
  <TabItem label="Python">
    Uvicorn serves the FastAPI application directly. The entry point receives its host and port from validated settings rather than treating the public URL as a listener address.

```python
import uvicorn

from order_service.config import settings


def main() -> None:
    uvicorn.run(
        "order_service.web:app",
        host="0.0.0.0",
        port=settings.port,
        access_log=False,
    )
```
  </TabItem>
  <TabItem label="Go">
    `http.Server` owns the listener and serves a chi router. `ListenAndServe` returns bind and runtime failures to the command entry point.

```go
package web

import (
    "errors"
    "fmt"
    "net/http"
    "strconv"

    "github.com/go-chi/chi/v5"

    "order-service/internal/config"
)

func Listen(cfg config.Config, orders http.HandlerFunc) error {
    router := chi.NewRouter()
    router.Get("/orders/{id}", orders)

    server := &http.Server{
        Addr:    ":" + strconv.Itoa(cfg.Port),
        Handler: router,
    }
    if err := server.ListenAndServe(); !errors.Is(err, http.ErrServerClosed) {
        return fmt.Errorf("serve HTTP: %w", err)
    }
    return nil
}
```
  </TabItem>
</Tabs>

**Evidence to collect:** Start the artifact with a new valid port and route traffic without installing a framework-specific server on the host.

## Factor VIII: Concurrency

Web and worker fleets scale as separate process types. Each worker also limits in-process concurrency so horizontal scaling does not multiply unbounded work.

<Tabs syncKey="twelve-factor-language">
  <TabItem label="TypeScript">
    BullMQ caps active jobs in each worker process. The platform can add worker replicas independently, while the per-process bound protects database and API capacity.

```typescript
import { Worker } from "bullmq";
import { Redis } from "ioredis";

import { config } from "./config.js";
import { fulfillOrder } from "./order-service.js";

type OrderJob = { orderId: string };

const connection = new Redis(config.REDIS_URL, {
  maxRetriesPerRequest: null,
});

export const worker = new Worker<OrderJob>(
  "orders",
  async (job) => fulfillOrder(job.data.orderId),
  {
    connection,
    concurrency: config.WORKER_CONCURRENCY,
    limiter: { max: 100, duration: 1_000 },
  },
);
```
  </TabItem>
  <TabItem label="Python">
    The Dramatiq actor defines retry and execution limits. The worker command fixes the process and thread counts, which gives each replica a known concurrency budget.

```python
import asyncio

import dramatiq

from order_service.orders import fulfill_order


@dramatiq.actor(queue_name="orders", max_retries=8, time_limit=30_000)
def fulfill_order_job(order_id: str) -> None:
    asyncio.run(fulfill_order(order_id))
```

```bash
dramatiq order_service.worker --processes 2 --threads 4
```
  </TabItem>
  <TabItem label="Go">
    Asynq gives the worker process a fixed concurrency and weighted queues. Adding replicas increases capacity in explicit increments instead of opening an unbounded goroutine path.

```go
package worker

import (
    "github.com/hibiken/asynq"

    "order-service/internal/config"
)

func Run(cfg config.Config, handler asynq.Handler) error {
    redis, err := asynq.ParseRedisURI(cfg.RedisURL)
    if err != nil {
        return err
    }
    server := asynq.NewServer(redis, asynq.Config{
        Concurrency: cfg.WorkerConcurrency,
        Queues: map[string]int{
            "critical": 3,
            "default":  1,
        },
    })
    return server.Run(handler)
}
```
  </TabItem>
</Tabs>

**Evidence to collect:** Increase worker replicas while holding the web fleet constant, then verify queue throughput improves without exceeding database connection limits.
